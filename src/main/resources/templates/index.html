2. JOIN FETCH (villas disponibles con tipo, proyecto y servicios)

SELECT DISTINCT v FROM Villa v JOIN FETCH v.tipo JOIN FETCH v.proyecto LEFT JOIN FETCH v.servicios WHERE v.estado = 'DISPONIBLE'
CodigoProyectoTipoServicios

3. UPDATE masivo (reajuste de precios)

UPDATE Villa v SET v.precio = v.precio * :factor WHERE v.proyecto.id = :proyectoId AND v.estado = 'DISPONIBLE'

4. JOIN FETCH de reservas

SELECT r FROM Reserva r JOIN FETCH r.villa JOIN FETCH r.cliente JOIN FETCH r.vendedor ORDER BY r.fecha

5. Agregacion GROUP BY

SELECT r.vendedor.nombre, COUNT(r), SUM(r.monto) FROM Reserva r GROUP BY r.vendedor.nombre
